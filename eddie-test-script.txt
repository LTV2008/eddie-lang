// Test 3: Control Flow
println("Test 3: Control Flow")
var a = 10
if (a < 20) {
    println("a is less than 20") // Output: a is less than 20
} else {
    println("a is 20 or more")
}

var b = 5
if (b >= 10) {
    println("b is 10 or more")
} else if (b > 2) {
    println("b is between 3 and 9") // Output: b is between 3 and 9
} else {
    println("b is 2 or less")
}

// Test 4: Arrays
println("Test 4: Arrays")
var arr = [1, 2, 3]
println(arr[0]) // Output: 1
arr[1] = 42
println(arr) // Output: [1, 42, 3]
arr.push(5)
println(arr) // Output: [1, 42, 3, 5]
println(arr.pop()) // Output: 5
println(arr) // Output: [1, 42, 3]
println(arr.length) // Output: 3
